BLOCO LETREIRO-VELOCIDADE - JS (id="letreiro-velocidade-js", no fim do <body>, depois do primeira-tela-js)
(function(){
  var PXS=69;
  function velocidade(){
    var g=document.querySelector('.mqg'),t=document.querySelector('.mqt');
    if(!g||!t)return;
    var lap=g.getBoundingClientRect().width;
    if(!lap)return;
    var d=(lap/PXS).toFixed(1)+'s';
    if(t.style.animationDuration!==d)t.style.animationDuration=d;
  }
  window.addEventListener('resize',velocidade);
  window.addEventListener('orientationchange',velocidade);
  window.addEventListener('load',velocidade);
  document.addEventListener('DOMContentLoaded',velocidade);
  if(document.fonts&&document.fonts.ready)document.fonts.ready.then(velocidade);
  velocidade();
})();

(function(){
  var lastW=0,lastH=0;
  function ajustar(force){
    var hero=document.querySelector('.ct-hero'),mq=document.querySelector('.mqw');
    if(!hero||!mq)return;
    var w=window.innerWidth,h=window.innerHeight;
    if(!force&&w<=980&&w===lastW&&Math.abs(h-lastH)<150)return;
    lastW=w;lastH=h;
    hero.style.minHeight='0px';
    var topo=hero.getBoundingClientRect().top+(window.pageYOffset||0);
    var free=h-topo-mq.offsetHeight;
    hero.style.minHeight=free>=200?free+'px':'';
  }
  function run(){ajustar(true);}
  window.addEventListener('resize',function(){ajustar(false);});
  window.addEventListener('orientationchange',run);
  window.addEventListener('load',run);
  document.addEventListener('DOMContentLoaded',run);
  if(document.fonts&&document.fonts.ready)document.fonts.ready.then(run);
  run();
})();
