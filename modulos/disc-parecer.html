⚙️ Conexão Supabase
A anon key fica em Supabase → Settings → API → "anon public". Começa sempre com eyJ
📋 SQL — Criar tabela pareceres_historico
create table if not exists pareceres_historico (
  id bigint generated always as identity primary key,
  cand_id text,
  nome text,
  vaga text,
  dom text,
  sec text,
  decisao text,
  parecer_json text,
  criado_em timestamptz default now(),
  atualizado_em timestamptz default now()
);
-- Habilitar RLS e permitir acesso anon:
alter table pareceres_historico enable row level security;
create policy "anon_all" on pareceres_historico for all to anon using (true) with check (true);
